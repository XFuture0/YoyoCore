<b><size=20>基本数学工具</size></b>

<i>数学是游戏开发的基石，而这些常用的数学计算，封装起来就不用每次都翻文档了。</i>

MathTool 提供了一系列游戏开发中常用的数学计算函数，包括角度转换、距离计算、射线检测等。

<b><size=16>核心功能</size></b>

<b>角度与弧度转换</b>

// 角度转弧度
float rad = MathTool.DegToRad(90f);  // 返回 π/2

// 弧度转角度
float deg = MathTool.RadToDeg(Mathf.PI);  // 返回 180

<b>平面距离计算</b>

在 3D 游戏中，我们经常需要忽略某个轴来计算距离：

Vector3 pos1 = new Vector3(1, 10, 2);
Vector3 pos2 = new Vector3(4, 5, 6);

// 忽略 Y 轴的距离（常用于地面移动计算）
float distXZ = MathTool.GetDistanceXZ(pos1, pos2);

// 忽略 Z 轴的距离
float distXY = MathTool.GetDistanceXY(pos1, pos2);

// 忽略 X 轴的距离
float distYZ = MathTool.GetDistanceYZ(pos1, pos2);

<b>距离检查</b>

Vector3 playerPos = player.transform.position;
Vector3 enemyPos = enemy.transform.position;

// 检查 XZ 平面距离是否小于 5
if (MathTool.CheckDistanceXZ(playerPos, enemyPos, 5f))
{
    Debug.Log("敌人进入攻击范围！");
}

<b>屏幕范围检测</b>

// 检查世界坐标是否在屏幕外
Vector3 worldPos = enemy.transform.position;
if (MathTool.IsWorldPosOutScreen(worldPos))
{
    Debug.Log("敌人在屏幕外");

<b>扇形范围检测</b>

常用于技能范围检测：

// 检查目标是否在扇形范围内
bool inRange = MathTool.IsInSectorRangeXZ(
    playerPos,      // 扇形中心位置
    playerForward,  // 扇形朝向
    targetPos,      // 目标位置
    5f,             // 扇形半径
    60f             // 扇形角度（总角度，每边 30 度）
);

<b>射线检测</b>

Ray ray = new Ray(transform.position, transform.forward);

// 检测并获取 RaycastHit
MathTool.Instance.RayCast(ray, (hit) =>
{
    Debug.Log($"命中: {hit.collider.name}");
}, 100f, LayerMask.GetMask("Enemy"));

// 检测并获取 GameObject
MathTool.Instance.RayCast(ray, (go) =>
{
    Debug.Log($"命中对象: {go.name}");
}, 100f, LayerMask.GetMask("Default"));

// 检测并获取组件
MathTool.Instance.RayCast<Enemy>(ray, (enemy) =>
{
    enemy.TakeDamage(10);
}, 100f, LayerMask.GetMask("Enemy"));

<b>技能范围检测</b>

public class Skill : MonoBehaviour
{
    public float range = 5f;
    public float angle = 60f;

    void CastSkill()
    {
        // 检测扇形范围内的敌人
        Collider[] enemies = Physics.OverlapSphere(transform.position, range);
        foreach (var enemy in enemies)
        {
            if (MathTool.IsInSectorRangeXZ(
                transform.position,
                transform.forward,
                enemy.transform.position,
                range,
                angle))
            {
                enemy.GetComponent<Enemy>().TakeDamage(100);
            }
        }
    }
}

<b>视野检测</b>

public class FieldOfView : MonoBehaviour
{
    public float viewRadius = 10f;
    public float viewAngle = 90f;

    void Update()
    {
        // 球形检测获取所有可能的目标
        MathTool.Instance.OverLapSphere<Collider>(
            transform.position,
            viewRadius,
            LayerMask.GetMask("Target"),
            (collider) =>
            {
                // 进一步检查是否在扇形视野内
                if (MathTool.IsInSectorRangeXZ(
                    transform.position,
                    transform.forward,
                    collider.transform.position,
                    viewRadius,
                    viewAngle))
                {
                    Debug.Log($"发现目标: {collider.name}");
                }
            }
        );
    }
}

<b><size=16>技术细节</size></b>

<b>平面距离计算原理</b>

public static float GetDistanceXZ(Vector3 pos1, Vector3 pos2)
{
    pos1.y = 0;  // 忽略 Y 轴
    pos2.y = 0;
    return Vector3.Distance(pos1, pos2);
}

<b>扇形检测原理</b>

public static bool IsInSectorRangeXZ(Vector3 pos, Vector3 forward, Vector3 target, float radius, float angle)
{
    pos.y = 0;
    target.y = 0;
    forward.y = 0;
    // 检查距离和角度
    return Vector3.Distance(pos, target) <= radius &&
           Vector3.Angle(forward, target - pos) <= angle / 2f;
}

<b><size=16>注意事项</size></b>

<size=13><b>1. 射线检测的 LayerMask</b>

使用射线检测时，正确设置 LayerMask 很重要：

// 只检测 Enemy 层
LayerMask.GetMask("Enemy")

// 检测多个层
LayerMask.GetMask("Enemy", "Obstacle")

// 排除特定层
~LayerMask.GetMask("UI")

<b>2. 性能考虑</b>

- OverlapSphere/OverlapBox 比 RaycastAll 更高效，优先使用
- 频繁的射线检测可能影响性能，考虑降低检测频率
- 使用非分配版本的物理检测（如 SphereCastNonAlloc）避免 GC

<b>3. 精度问题</b>

浮点数计算可能有精度误差：

// 比较距离时考虑误差
if (dist <= range + 0.001f)
{
    // 在范围内
}</c></size>

<b><size=16>技术地位和意义</size></b>

MathTool 是框架中负责「数学计算」的工具集。它将常用的数学操作封装起来，让开发者可以更专注于游戏逻辑而不是数学细节。

<i>毕竟，游戏开发中 80% 的数学都是「距离、角度、射线」这三板斧，封装好了能省不少事。</i>
